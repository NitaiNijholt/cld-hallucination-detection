\begin{tikzpicture}[

    data_label/.style={font=\scriptsize, text=gray!50, align=center},

% Column positions - calculated for equal 0.8cm gaps between box edges
% Phase 2 is wider (6.36cm) due to split branches at ±1.5, others are 4.2cm
\def\colGen{0}
\def\colJudge{6.1}

% ==========================================
% PHASE 1: GENERATION (Green) - Column 1
% ==========================================

\node[io, io_style] (input) at (\colGen, 0) {Context Parameters\\{\tiny target, temporal, spatial}};
\node[process, gen_color] (node_gen) at (\colGen, -1.2) {\texttt{generate\_variables}\\{\tiny GPT-4.1}};
\node[process, gen_color] (edge_gen) at (\colGen, -2.4) {\texttt{discover\_relationships}\\{\tiny GPT-4.1, $N^2$ pairs}};
\node[process, gen_color] (uq_metrics) at (\colGen, -3.6) {\texttt{compute\_uq\_metrics}\\{\tiny PPL, $P_{min}$, $H_{win}$, cosine sim}};
\node[io, io_style] (gen_cld) at (\colGen, -4.8) {Generated CLD\\{\tiny nodes, edges, citations}};

% ==========================================
% PHASE 1.1: CORRUPTION (Gray/Dashed) - Below Generation Phase
% ==========================================
\node[process, corrupt_style, minimum width=2.4cm] (corruptor) at (\colGen, -6.6) {\texttt{corruptor}\\{\tiny GT Synth}};

% ==========================================
% PHASE 2: JUDGING (Blue) - Column 2
% ==========================================

\node[process, judge_color] (judge) at (\colJudge, 0) {\texttt{judge\_all\_edges}\\{\tiny GPT-4.1}};
\node[decision, judge_color] (judge_type) at (\colJudge, -1.3) {Approach?};
\node[process, judge_color] (judge_corr) at (\colJudge-1.5, -2.6) {\texttt{judgeCorrectness}\\{\tiny reasoning quality}};
\node[process, judge_color] (judge_cit) at (\colJudge+1.5, -2.6) {\texttt{judgeCitation}\\{\tiny fetch \& verify}};
\node[process, judge_color] (aggregate) at (\colJudge, -3.8) {Aggregate Verdicts\\{\tiny majority vote}};
\node[io, io_style] (judged_cld) at (\colJudge, -5.0) {Judged CLD\\{\tiny verdict, score per edge}};

% ==========================================
% PHASE 3: CORRECTION (Orange) - Column 3
% ==========================================

\node[process, correct_color] (filter) at (\colCorrect, 0) {Filter Candidates\\{\tiny INCORRECT, PARTIAL}};
\node[process, correct_color] (corrector) at (\colCorrect, -1.2) {\texttt{corrector\_agent}\\{\tiny GPT-4.1 + tools}};
\node[process, correct_color] (rejudge) at (\colCorrect, -2.4) {Re-judge Corrected Edges\\{\tiny same judge pipeline}};
\node[io, io_style] (corrected_cld) at (\colCorrect, -5.0) {Corrected CLD\\{\tiny updated edges, logs}};

% ==========================================
% PHASE 4: DEEP RESEARCH (Purple) - Column 4
% ==========================================

\node[process, dr_color] (dr_sample) at (\colDR, 0) {Sample Edges\\{\tiny stratified by TP/FP/FN}};
\node[process, dr_color] (dr_search) at (\colDR, -1.2) {\texttt{deep\_research}\\{\tiny GPT-5 + web search}};
\node[process, dr_color] (dr_evidence) at (\colDR, -2.4) {Evidence Detection\\{\tiny direct causal support}};
\node[process, dr_color] (dr_synthesize) at (\colDR, -3.6) {Synthesize Verdict\\{\tiny confidence, sources}};
\node[io, io_style] (dr_output) at (\colDR, -5.0) {Validated Edges\\{\tiny evidence, URLs, verdict}};

    -- ++(0, 1.0)
    -- ++(-1.2, 0)
    |- (judge.east);

\draw[arrow] (rejudge.south) -- (corrected_cld.north);

% ARROWS - Deep Research Phase
% ==========================================

\draw[arrow] (dr_search) -- (dr_evidence);
