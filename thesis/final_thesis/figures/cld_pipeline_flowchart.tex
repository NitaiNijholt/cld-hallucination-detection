% CLD Pipeline Flowchart: Generation → Judging → Correction
% Color-coded by phase: Generation (green), Judging (blue), Correction (orange)
% Horizontal layout with phases as columns
\begin{tikzpicture}[
    % === NODE STYLES ===
    gen_color/.style={fill=green!15, draw=green!60!black, line width=0.7pt},
    judge_color/.style={fill=blue!15, draw=blue!60!black, line width=0.7pt},
    correct_color/.style={fill=orange!20, draw=orange!70!black, line width=0.7pt},
    io_style/.style={fill=gray!10, draw=gray!60, line width=0.7pt, rounded corners=3pt},
    decision/.style={diamond, aspect=2.2, minimum width=1.4cm, minimum height=0.6cm, text width=1.5cm, align=center, font=\scriptsize},
    process/.style={rectangle, rounded corners=3pt, minimum width=2.8cm, minimum height=0.7cm, align=center, font=\scriptsize},
    io/.style={rectangle, rounded corners=3pt, minimum width=2.6cm, minimum height=0.6cm, align=center, font=\scriptsize},
    arrow/.style={-{Stealth[length=2.2mm, width=1.6mm]}, line width=0.5pt, draw=gray!70},
    label_style/.style={font=\scriptsize, text=gray!60},
    phase_label/.style={font=\small\bfseries, align=center},
    data_label/.style={font=\scriptsize, text=gray!50, align=center},
]

% Column positions - more spread out
\def\colGen{0}
\def\colJudge{5.5}
\def\colCorrect{11}

% ==========================================
% PHASE 1: GENERATION (Green) - Column 1
% ==========================================

\node[io, io_style] (input) at (\colGen, 0) {Context Parameters\\{\tiny target, temporal, spatial}};
\node[process, gen_color] (node_gen) at (\colGen, -1.2) {\texttt{generate\_variables}\\{\tiny GPT-4.1}};
\node[process, gen_color] (edge_gen) at (\colGen, -2.4) {\texttt{discover\_relationships}\\{\tiny GPT-4.1, $N^2$ pairs}};
\node[process, gen_color] (uq_metrics) at (\colGen, -3.6) {\texttt{compute\_uq\_metrics}\\{\tiny PPL, $P_{min}$, $H_{win}$, cosine sim}};
\node[io, io_style] (gen_cld) at (\colGen, -4.8) {Generated CLD\\{\tiny nodes, edges, citations}};

% ==========================================
% PHASE 2: JUDGING (Blue) - Column 2
% ==========================================

\node[process, judge_color] (judge) at (\colJudge, 0) {\texttt{judge\_all\_edges}\\{\tiny GPT-4.1}};
\node[decision, judge_color] (judge_type) at (\colJudge, -1.3) {Approach?};
\node[process, judge_color] (judge_corr) at (\colJudge-1.6, -2.6) {\texttt{judgeCorrectness}\\{\tiny reasoning quality}};
\node[process, judge_color] (judge_cit) at (\colJudge+1.6, -2.6) {\texttt{judgeCitation}\\{\tiny fetch \& verify}};
\node[process, judge_color] (aggregate) at (\colJudge, -3.8) {Aggregate Verdicts\\{\tiny majority vote}};
\node[io, io_style] (judged_cld) at (\colJudge, -5.0) {Judged CLD\\{\tiny verdict, score per edge}};

% ==========================================
% PHASE 3: CORRECTION (Orange) - Column 3
% ==========================================

\node[process, correct_color] (filter) at (\colCorrect, 0) {Filter Candidates\\{\tiny INCORRECT, PARTIAL}};
\node[process, correct_color] (corrector) at (\colCorrect, -1.2) {\texttt{corrector\_agent}\\{\tiny GPT-4.1 + tools}};
\node[process, correct_color] (rejudge) at (\colCorrect, -2.4) {Re-judge All Edges\\{\tiny same judge pipeline}};
\node[io, io_style] (corrected_cld) at (\colCorrect, -5.0) {Corrected CLD\\{\tiny updated edges, logs}};

% ==========================================
% ARROWS - Generation Phase
% ==========================================
\draw[arrow] (input) -- (node_gen);
\draw[arrow] (node_gen) -- (edge_gen);
\draw[arrow, dashed] (node_gen.east) -- ++(0.4,0) |- (uq_metrics.east);
\draw[arrow] (edge_gen) -- (uq_metrics);
\draw[arrow] (uq_metrics) -- (gen_cld);

% ==========================================
% ARROWS - Transition Gen → Judge
% ==========================================
\draw[arrow] (gen_cld.east) -- ++(0.6, 0) -- ++(0, 4.3) -- (\colJudge-1.5, -0.5) -- (\colJudge-1.5, 0) -- (judge.west);

% ==========================================
% ARROWS - Judging Phase
% ==========================================
\draw[arrow] (judge) -- (judge_type);
\draw[arrow] (judge_type.west) -| (judge_corr.north);
\node[label_style] at (\colJudge-1.1, -1.7) {Corr.};
\draw[arrow] (judge_type.east) -| (judge_cit.north);
\node[label_style] at (\colJudge+1.1, -1.7) {Cit.};
\draw[arrow] (judge_corr.south) |- (aggregate.west);
\draw[arrow] (judge_cit.south) |- (aggregate.east);
\draw[arrow] (aggregate) -- (judged_cld);

% ==========================================
% ARROWS - Transition Judge → Correct
% ==========================================
% Draw this transition *after* `correct_box` exists, so we can anchor the arrowhead
% exactly on the Phase 3 box border (see below).

% ==========================================
% ARROWS - Correction Phase
% ==========================================
\draw[arrow] (filter) -- (corrector);
\draw[arrow] (corrector) -- (rejudge);

% Re-judge loops back to judge
% Route the loop ABOVE the phase boxes to avoid overlapping the Judging internals.
\draw[arrow]
    (rejudge.west) -- ++(-1.0, 0)
    |- (\colCorrect-2.6, 1.25)
    -- (\colJudge+2.6, 1.25)
    -| (judge.east);

% Re-judge also outputs to corrected CLD
\draw[arrow] (rejudge.south) -- (corrected_cld.north);

% ==========================================
% PHASE LABELS (Background boxes)
% ==========================================
\begin{scope}[on background layer]
    \node[draw=green!60!black, fill=green!5, rounded corners=5pt, 
          fit=(input)(node_gen)(edge_gen)(uq_metrics)(gen_cld), 
          inner sep=8pt] (gen_box) {};
    \node[phase_label, text=green!60!black] at (\colGen, 0.9) {\textbf{Phase 1: Generation}};
    
    \node[draw=blue!60!black, fill=blue!5, rounded corners=5pt,
          fit=(judge)(judge_type)(judge_corr)(judge_cit)(aggregate)(judged_cld),
          inner sep=8pt] (judge_box) {};
    \node[phase_label, text=blue!60!black] at (\colJudge, 0.9) {\textbf{Phase 2: Judging}};
    
    \node[draw=orange!70!black, fill=orange!5, rounded corners=5pt,
          fit=(filter)(corrector)(rejudge)(corrected_cld),
          inner sep=8pt] (correct_box) {};
    \node[phase_label, text=orange!70!black] at (\colCorrect, 0.9) {\textbf{Phase 3: Correction}};
\end{scope}

% ==========================================
% ARROWS - Transition Judge → Correct (arrowhead on Phase 3 box edge)
% ==========================================
\coordinate (phase3_edge) at (correct_box.west |- filter.west);
\draw[arrow]
    (judged_cld.east) -- ++(0.6, 0)
    -- ++(0, 4.5)
    -- (\colCorrect-2.6, 1.25)
    |- (phase3_edge);
\draw[line width=0.5pt, draw=gray!70] (phase3_edge) -- (filter.west);

\end{tikzpicture}
